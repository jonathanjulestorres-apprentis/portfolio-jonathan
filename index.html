1. Régulation Thermique par PWM (Projet TIPE)

Arduino UNO C/C++ PWM I2C Acquisition de données
Contexte et Objectif : L'objectif de ce projet est de concevoir un système de ventilation intelligent, similaire à un ventirad de PC. Au lieu d'avoir un ventilateur qui tourne constamment à pleine vitesse, le système détecte la chaleur du processeur (ou de la résistance) et ajuste automatiquement sa vitesse de rotation proportionnellement à la hausse de température.

Fonctionnement : Le système s'appuie sur trois capteurs DHT11 pour cartographier la chaleur (extérieur, source, boîtier). Une boucle fermée gère l'asservissement : si la température dépasse 45°C, le ventilateur s'active et sa vitesse augmente par paliers, via une commande PWM.

Intégration du système dans un boîtier PC pour tests réels avec source chauffante

Montage électronique (Arduino, écran LCD I2C, câblage)

C++ / Code Arduino Complet (Scrollez pour lire)

#include <DHT.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// ======================================================
// CONFIGURATION
// ======================================================

#define DHTTYPE DHT11

// DHT11
#define DHT_EXT 5       // Température extérieure
#define DHT_SRC 6       // Température près de la résistance
#define DHT_BOX 7       // Température dans le boîtier

// Ventilateur 4 broches
#define FAN_PWM 9       // Fil bleu : PWM
#define TACH_PIN 2      // Fil jaune : tachymètre

// LCD I2C
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Capteurs
DHT dhtExt(DHT_EXT, DHTTYPE);
DHT dhtSrc(DHT_SRC, DHTTYPE);
DHT dhtBox(DHT_BOX, DHTTYPE);

// Compteur d'impulsions du tachymètre
volatile unsigned long pulseCount = 0;

// ======================================================
// INTERRUPTION DU TACHYMETRE
// ======================================================

void countPulse()
{
  pulseCount++;
}

// ======================================================
// SETUP
// ======================================================

void setup()
{
  Serial.begin(9600);

  // Initialisation des DHT11
  dhtExt.begin();
  dhtSrc.begin();
  dhtBox.begin();

  // PWM du ventilateur
  pinMode(FAN_PWM, OUTPUT);

  // Tachymètre
  pinMode(TACH_PIN, INPUT_PULLUP);

  // Interruption sur le tachymètre
  attachInterrupt(
    digitalPinToInterrupt(TACH_PIN),
    countPulse,
    FALLING
  );

  // LCD
  lcd.init();
  lcd.backlight();

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Systeme thermique");

  lcd.setCursor(0, 1);
  lcd.print("Initialisation");

  delay(2000);
  lcd.clear();

  // En-tête pour Excel
  Serial.println(
    "Temps(s);"
    "Text(C);"
    "Tsource(C);"
    "Tboitier(C);"
    "Veq(V);"
    "PWM;"
    "RPM"
  );
}

// ======================================================
// LOOP
// ======================================================

void loop()
{
  // 1. LECTURE DES TEMPERATURES
  float Text = dhtExt.readTemperature();
  float Tsrc = dhtSrc.readTemperature();
  float Tbox = dhtBox.readTemperature();

  if (isnan(Text) || isnan(Tsrc) || isnan(Tbox))
  {
    Serial.println("ERREUR : lecture d'un DHT11 impossible");
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Erreur capteur");
    delay(2000);
    return;
  }

  // 2. DETERMINATION DE LA COMMANDE DU VENTILATEUR
  int pwm = 0;
  int tensionEquivalent = 0;

  if (Tbox < 45) {
    pwm = 0;
    tensionEquivalent = 0;
  }
  else if (Tbox < 50) {
    pwm = 170;
    tensionEquivalent = 8;
  }
  else if (Tbox < 55) {
    pwm = 191;
    tensionEquivalent = 9;
  }
  else if (Tbox < 65) {
    pwm = 212;
    tensionEquivalent = 10;
  }
  else if (Tbox < 75) {
    pwm = 234;
    tensionEquivalent = 11;
  }
  else {
    pwm = 255;
    tensionEquivalent = 12;
  }

  // Application du PWM
  analogWrite(FAN_PWM, pwm);

  // 3. MESURE DU TACHYMETRE
  noInterrupts();
  pulseCount = 0;
  interrupts();

  unsigned long debutMesure = millis();
  while (millis() - debutMesure < 1000) {
    // Les impulsions sont comptées automatiquement
  }

  noInterrupts();
  unsigned long pulses = pulseCount;
  interrupts();

  // 4. CALCUL DES RPM
  float rpm = pulses * 30.0;

  // 5. AFFICHAGE LCD
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("T:");
  lcd.print(Tbox, 1);
  lcd.print((char)223);
  lcd.print("C");
  lcd.setCursor(9, 0);
  lcd.print("RPM");

  lcd.setCursor(0, 1);
  lcd.print("Fan:");
  lcd.print(tensionEquivalent);
  lcd.print("V");
  lcd.setCursor(9, 1);
  lcd.print((int)rpm);

  // 6. EXPORT DES DONNEES POUR EXCEL
  float temps = millis() / 1000.0;
  Serial.print(temps); Serial.print(";");
  Serial.print(Text, 1); Serial.print(";");
  Serial.print(Tsrc, 1); Serial.print(";");
  Serial.print(Tbox, 1); Serial.print(";");
  Serial.print(tensionEquivalent); Serial.print(";");
  Serial.print(pwm); Serial.print(";");
  Serial.println(rpm);

  delay(1000);
}

2. Tracker Solaire Automatique (Projet TIPE)

Arduino Servomoteur SG90 Photorésistances (LDR) Logique différentielle
Contexte et Objectif : Un panneau solaire fixe perd énormément de rendement car il n'est bien orienté que quelques heures par jour. Ce projet consiste à concevoir un "Tracker" : un support motorisé intelligent qui suit la course du soleil, du lever au coucher, en s'orientant toujours vers la zone la plus lumineuse afin d'emmagasiner le maximum d'énergie possible.

Fonctionnement : Deux photorésistances (LDR) sont placées de chaque côté du panneau et mesurent la luminosité ambiante. L'Arduino compare en permanence ces deux valeurs. S'il détecte que la lumière est plus forte à gauche, il actionne le servomoteur pour faire pivoter le panneau vers la gauche, et inversement, jusqu'à équilibre.

Simulation complète du circuit et de l'algorithme sur Tinkercad

Voir la simulation interactive Tinkercad
C++ / Code Arduino Complet

#include <Servo.h>

Servo myServo;
int iRotationServo = 90; // Position initiale au centre

int iPhotoValue1 = 0;
int myPin1 = A0; // LDR gauche

int iPhotoValue2 = 0;
int myPin2 = A1; // LDR droite

void setup() {
  Serial.begin(9600);
  myServo.attach(8);
  myServo.write(iRotationServo);
  delay(500);
}

void loop() {
  // 1. Lecture de la luminosité
  iPhotoValue1 = analogRead(myPin1);
  iPhotoValue2 = analogRead(myPin2);

  // 2. Logique d'asservissement différentiel
  if (iPhotoValue1 > iPhotoValue2) {
    iRotationServo++; // Plus de lumière à gauche
    if (iRotationServo > 180) iRotationServo = 180;
    myServo.write(iRotationServo);
  }
  else if (iPhotoValue1 < iPhotoValue2) {
    iRotationServo--; // Plus de lumière à droite
    if (iRotationServo < 0) iRotationServo = 0;
    myServo.write(iRotationServo);
  }

  delay(100); // Lissage du mouvement
}

3. Carrefour Intelligent et Véhicule Autonome (Projet Personnel)

Arduino Mega Capteur Couleur (TCS3200) Ultrasons (HC-SR04) Impression 3D / CAO
Contexte et Objectif : Ce projet, toujours en cours de réalisation au FabLab, simule un environnement routier intelligent. L'objectif est de développer une voiture autonome capable de pallier les défaillances humaines (conducteur inattentif ou en incapacité). Le système intègre un passage à niveau automatisé et un feu tricolore interactif : le véhicule détecte la couleur du feu et les obstacles, enclenchant un freinage d'urgence automatique si nécessaire.

Fonctionnement : La maquette met en jeu de nombreux actionneurs et capteurs : la voiture embarque un capteur colorimétrique (pour "lire" l'état du feu) et un capteur à ultrasons. Une machine à états gère la prise de décision : si le feu passe au rouge à l'approche d'un train, ou si un obstacle est trop proche, un signal PWM commande un freinage progressif et immédiat des moteurs (via driver L298N).

Maquette fabriquée au FabLab (bois, découpe laser, impression 3D) intégrant le véhicule, la voie ferrée et la barrière automatisée.
